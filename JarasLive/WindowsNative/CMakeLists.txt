cmake_minimum_required(VERSION 3.24)
project(CatLiveNativeCore VERSION 1.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
set(CATLIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(CatLiveCore STATIC
    "${CATLIVE_ROOT}/Core/Project/Models.cpp"
    "${CATLIVE_ROOT}/Core/Transport/Engine.cpp"
    "${CATLIVE_ROOT}/Core/Import/TrackTaxonomy.cpp")
target_include_directories(CatLiveCore PUBLIC "${CATLIVE_ROOT}/Core")
add_library(CatLiveDocuments STATIC Source/ProjectCodec.cpp Source/NativeSession.cpp Source/SessionActions.cpp Source/NativeBridge.cpp Source/TrackDuplication.cpp Source/ProjectCreation.cpp)
target_include_directories(CatLiveDocuments PUBLIC Source ThirdParty)
target_link_libraries(CatLiveDocuments PUBLIC CatLiveCore)
if(WIN32)
    target_link_libraries(CatLiveDocuments PRIVATE bcrypt normaliz)
endif()
add_library(CatLiveSoundFont STATIC Source/SoundFontRenderer.cpp)
target_include_directories(CatLiveSoundFont PUBLIC Source)
add_library(CatLiveAudioFiles STATIC Source/AudioFile.cpp)
target_include_directories(CatLiveAudioFiles PUBLIC Source)
if(MSVC)
    target_compile_options(CatLiveSoundFont PRIVATE /utf-8 /W4)
endif()
if(MSVC)
    target_compile_options(CatLiveCore PRIVATE /utf-8 /W4)
    target_compile_options(CatLiveDocuments PRIVATE /utf-8 /W4 /bigobj)
endif()
enable_testing()
if(WIN32)
    add_library(CatLiveVST3 STATIC Source/VST3Host.cpp
        "${CATLIVE_ROOT}/Core/ThirdParty/VST3/pluginterfaces/base/funknown.cpp"
        "${CATLIVE_ROOT}/Core/ThirdParty/VST3/pluginterfaces/base/ustring.cpp"
        "${CATLIVE_ROOT}/Core/ThirdParty/VST3/pluginterfaces/base/coreiids.cpp")
    target_include_directories(CatLiveVST3 PUBLIC Source PRIVATE "${CATLIVE_ROOT}/Core/ThirdParty/VST3")
    target_compile_definitions(CatLiveVST3 PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(CatLiveVST3 PRIVATE ole32 user32)
    add_executable(VST3Probe Tests/VST3Probe.cpp)
    target_include_directories(VST3Probe PRIVATE ThirdParty)
    target_link_libraries(VST3Probe PRIVATE CatLiveVST3)
endif()
foreach(name Core PlaybackScope SmoothSeek MultiLoop RegionBoundaryOwnership RegionOnsetOwnership TempoOwnership Timecode TimecodePlayback ClickTrack)
    add_executable(${name}Tests "${CATLIVE_ROOT}/Tests/Core/${name}Tests.cpp")
    target_link_libraries(${name}Tests PRIVATE CatLiveCore)
    add_test(NAME ${name}Tests COMMAND ${name}Tests)
endforeach()
add_executable(ProjectCodecTests Tests/ProjectCodecTests.cpp)
target_link_libraries(ProjectCodecTests PRIVATE CatLiveDocuments)
add_test(NAME ProjectCodecTests COMMAND ProjectCodecTests)
add_executable(PlaylistOccurrencesTests Tests/PlaylistOccurrencesTests.cpp)
target_link_libraries(PlaylistOccurrencesTests PRIVATE CatLiveDocuments)
add_test(NAME PlaylistOccurrencesTests COMMAND PlaylistOccurrencesTests)
add_executable(NativeSessionTests Tests/NativeSessionTests.cpp)
target_link_libraries(NativeSessionTests PRIVATE CatLiveDocuments)
add_test(NAME NativeSessionTests COMMAND NativeSessionTests)
add_executable(SessionActionsTests Tests/SessionActionsTests.cpp)
target_link_libraries(SessionActionsTests PRIVATE CatLiveDocuments)
add_test(NAME SessionActionsTests COMMAND SessionActionsTests)
add_executable(NativeBridgeTests Tests/NativeBridgeTests.cpp)
target_link_libraries(NativeBridgeTests PRIVATE CatLiveDocuments)
add_test(NAME NativeBridgeTests COMMAND NativeBridgeTests)
add_executable(TrackDuplicationTests Tests/TrackDuplicationTests.cpp)
target_link_libraries(TrackDuplicationTests PRIVATE CatLiveDocuments)
add_test(NAME TrackDuplicationTests COMMAND TrackDuplicationTests)
add_executable(ProjectCreationTests Tests/ProjectCreationTests.cpp)
target_link_libraries(ProjectCreationTests PRIVATE CatLiveDocuments)
add_test(NAME ProjectCreationTests COMMAND ProjectCreationTests)
add_executable(AudioFileTests Tests/AudioFileTests.cpp)
target_link_libraries(AudioFileTests PRIVATE CatLiveAudioFiles)
set(CATLIVE_TEST_MP3 "" CACHE FILEPATH "External MP3 fixture for sample-accurate seek tests (not distributed)")
if(CATLIVE_TEST_MP3)
    add_test(NAME AudioFileTests COMMAND AudioFileTests "${CATLIVE_TEST_MP3}")
else()
    add_test(NAME AudioFileTests COMMAND AudioFileTests)
endif()
set(CATLIVE_TEST_SF2 "" CACHE FILEPATH "External SF2 fixture for renderer tests (not distributed)")
if(CATLIVE_TEST_SF2)
    add_executable(SoundFontTests Tests/SoundFontTests.cpp Source/SoundFontRenderer.cpp)
    target_include_directories(SoundFontTests PRIVATE Source)
    target_compile_definitions(SoundFontTests PRIVATE CATLIVE_SOUNDFONT_TEST_ALLOCATOR)
    add_test(NAME SoundFontTests COMMAND SoundFontTests "${CATLIVE_TEST_SF2}")
endif()
