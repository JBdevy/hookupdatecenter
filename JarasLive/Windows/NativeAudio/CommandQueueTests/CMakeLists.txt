cmake_minimum_required(VERSION 3.20)
project(CatLiveCommandQueueTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(command_queue_tests main.cpp)
target_compile_features(command_queue_tests PRIVATE cxx_std_17)
target_link_libraries(command_queue_tests PRIVATE Threads::Threads)
option(COMMAND_QUEUE_SANITIZE "Address and undefined behavior sanitizers" OFF)
option(COMMAND_QUEUE_THREAD_SANITIZE "Thread sanitizer (separate build)" OFF)
if(COMMAND_QUEUE_SANITIZE AND COMMAND_QUEUE_THREAD_SANITIZE)
    message(FATAL_ERROR "Use separate sanitizer builds")
endif()
if(MSVC)
    target_compile_options(command_queue_tests PRIVATE /W4 /utf-8)
else()
    target_compile_options(command_queue_tests PRIVATE -Wall -Wextra -Wpedantic)
    if(COMMAND_QUEUE_SANITIZE)
        target_compile_options(command_queue_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(command_queue_tests PRIVATE -fsanitize=address,undefined)
    endif()
    if(COMMAND_QUEUE_THREAD_SANITIZE)
        target_compile_options(command_queue_tests PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
        target_link_options(command_queue_tests PRIVATE -fsanitize=thread)
    endif()
endif()
add_test(NAME command_queue COMMAND command_queue_tests)
set_tests_properties(command_queue PROPERTIES TIMEOUT 30)
