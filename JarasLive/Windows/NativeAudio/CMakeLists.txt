cmake_minimum_required(VERSION 3.24)
project(CatLiveNativeAudio LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The native device backends require Windows")
endif()
add_library(CatLiveNativeAudio STATIC WasapiDevice.cpp CallbackHub.cpp ClockedDuplexDevice.cpp)
target_compile_features(CatLiveNativeAudio PUBLIC cxx_std_20)
target_include_directories(CatLiveNativeAudio PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(CatLiveNativeAudio PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(CatLiveNativeAudio PRIVATE avrt ole32 uuid propsys winmm)
if(MSVC)
  target_compile_options(CatLiveNativeAudio PRIVATE /W4 /utf-8 /permissive-)
endif()
# The SDK stays outside the source tree. Never download or redistribute it as
# part of a release build; distribution requires the applicable SDK agreement.
set(CATLIVE_ASIO_SDK "" CACHE PATH "Externally supplied Steinberg ASIO SDK root")
if(CATLIVE_ASIO_SDK)
  if(NOT EXISTS "${CATLIVE_ASIO_SDK}/common/iasiodrv.h")
    message(FATAL_ERROR "CATLIVE_ASIO_SDK must point to an ASIO SDK with common/iasiodrv.h")
  endif()
  target_sources(CatLiveNativeAudio PRIVATE AsioDevice.cpp)
  target_include_directories(CatLiveNativeAudio PRIVATE "${CATLIVE_ASIO_SDK}/common")
  target_compile_definitions(CatLiveNativeAudio PUBLIC CATLIVE_DIRECT_ASIO=1 PRIVATE CATLIVE_HAS_ASIO_SDK=1)
  target_link_libraries(CatLiveNativeAudio PRIVATE advapi32)
endif()
add_executable(CatLiveNativeAudioProbe AudioDeviceProbe.cpp CallbackHubTests/AllocationAudit.cpp)
target_link_libraries(CatLiveNativeAudioProbe PRIVATE CatLiveNativeAudio)
if(MSVC)
  target_compile_options(CatLiveNativeAudioProbe PRIVATE /W4 /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(CatLivePCMConversionTests PCMConversionTests.cpp)
  add_executable(CatLiveASIOConversionTests AsioSampleConversionTests.cpp AsioDevice.cpp)
  add_executable(CatLiveCallbackHubTests CallbackHubTests/main.cpp CallbackHub.cpp)
  add_executable(CatLiveCommandQueueTests CommandQueueTests/main.cpp)
  add_executable(CatLiveDuplexInputTests DuplexTests/main.cpp CallbackHubTests/AllocationAudit.cpp)
  add_executable(CatLiveDuplexDeviceTests DuplexTests/DeviceTests.cpp ClockedDuplexDevice.cpp CallbackHubTests/AllocationAudit.cpp)
  set(catlive_native_audio_tests CatLivePCMConversionTests CatLiveASIOConversionTests CatLiveCallbackHubTests CatLiveCommandQueueTests CatLiveDuplexInputTests CatLiveDuplexDeviceTests)
  if(CATLIVE_ASIO_SDK)
    # Compiles the production ASIO implementation through a test-only boundary;
    # no registry changes, registered driver DLL, or hardware are involved.
    add_executable(CatLiveASIOLifecycleTests AsioLifecycleTests.cpp)
    target_include_directories(CatLiveASIOLifecycleTests PRIVATE "${CATLIVE_ASIO_SDK}/common")
    target_compile_definitions(CatLiveASIOLifecycleTests PRIVATE CATLIVE_HAS_ASIO_SDK=1 NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(CatLiveASIOLifecycleTests PRIVATE ole32)
    list(APPEND catlive_native_audio_tests CatLiveASIOLifecycleTests)
  endif()
  foreach(test IN LISTS catlive_native_audio_tests)
    target_compile_features(${test} PRIVATE cxx_std_20)
    if(MSVC)
      target_compile_options(${test} PRIVATE /W4 /utf-8 /UNDEBUG)
    else()
      target_compile_options(${test} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${test} COMMAND ${test})
    set_tests_properties(${test} PROPERTIES TIMEOUT 30)
  endforeach()
  set_tests_properties(CatLiveDuplexInputTests PROPERTIES TIMEOUT 180)
endif()
