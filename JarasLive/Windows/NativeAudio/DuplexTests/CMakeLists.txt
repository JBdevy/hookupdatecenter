cmake_minimum_required(VERSION 3.20)
project(CatLiveDuplexInputTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(duplex_input_tests main.cpp ../CallbackHubTests/AllocationAudit.cpp)
add_executable(duplex_device_tests DeviceTests.cpp ../ClockedDuplexDevice.cpp ../CallbackHubTests/AllocationAudit.cpp)
option(DUPLEX_SANITIZE "ASan and UBSan" OFF)
option(DUPLEX_THREAD_SANITIZE "TSan (separate build)" OFF)
if(DUPLEX_SANITIZE AND DUPLEX_THREAD_SANITIZE)
 message(FATAL_ERROR "Use separate sanitizer builds")
endif()
foreach(test IN ITEMS duplex_input_tests duplex_device_tests)
 target_compile_features(${test} PRIVATE cxx_std_20)
 target_link_libraries(${test} PRIVATE Threads::Threads)
 if(MSVC)
 target_compile_options(${test} PRIVATE /W4 /utf-8)
 else()
 target_compile_options(${test} PRIVATE -Wall -Wextra -Wpedantic)
 foreach(kind IN ITEMS SANITIZE THREAD_SANITIZE)
  if(DUPLEX_${kind})
   if(kind STREQUAL "SANITIZE")
    set(flag -fsanitize=address,undefined)
   else()
    set(flag -fsanitize=thread)
   endif()
   target_compile_options(${test} PRIVATE ${flag} -fno-omit-frame-pointer)
   target_link_options(${test} PRIVATE ${flag})
  endif()
 endforeach()
 endif()
endforeach()
add_test(NAME duplex_input COMMAND duplex_input_tests)
add_test(NAME duplex_device COMMAND duplex_device_tests)
