cmake_minimum_required(VERSION 3.20)
project(CatLiveNativeCallbackHubTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(callback_hub_tests main.cpp ../CallbackHub.cpp)
target_compile_features(callback_hub_tests PRIVATE cxx_std_20)
target_link_libraries(callback_hub_tests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(callback_hub_tests PRIVATE /W4 /utf-8)
else()
    target_compile_options(callback_hub_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(CALLBACK_HUB_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(CALLBACK_HUB_THREAD_SANITIZE "Enable thread sanitizer (separate build)" OFF)
if(CALLBACK_HUB_SANITIZE AND CALLBACK_HUB_THREAD_SANITIZE)
    message(FATAL_ERROR "Use separate builds for address and thread sanitizers")
endif()
if(NOT MSVC AND CALLBACK_HUB_SANITIZE)
    target_compile_options(callback_hub_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(callback_hub_tests PRIVATE -fsanitize=address,undefined)
endif()
if(NOT MSVC AND CALLBACK_HUB_THREAD_SANITIZE)
    target_compile_options(callback_hub_tests PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(callback_hub_tests PRIVATE -fsanitize=thread)
endif()
add_test(NAME native_callback_hub COMMAND callback_hub_tests)
