cmake_minimum_required(VERSION 3.24)
project(CatLiveIntegration VERSION 1.28.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0)
endif()
find_package(Threads REQUIRED)
add_library(IntegrationCore STATIC Source/Bridge.cpp Source/Resolume.cpp Source/GrandMA2.cpp Source/OSC.cpp)
target_include_directories(IntegrationCore PUBLIC Source ../JarasLive/WindowsNative/ThirdParty PRIVATE ThirdParty/httplib)
target_link_libraries(IntegrationCore PUBLIC Threads::Threads)
if(WIN32)
  enable_language(RC)
  set_property(TARGET IntegrationCore PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_sources(IntegrationCore PRIVATE Source/PlatformWindows.cpp)
  target_compile_definitions(IntegrationCore PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_compile_options(IntegrationCore PUBLIC /utf-8)
  target_link_libraries(IntegrationCore PUBLIC ws2_32 iphlpapi winmm bcrypt shell32 ole32)
  configure_file(Resources/Windows.rc.in "${CMAKE_CURRENT_BINARY_DIR}/Windows.rc" @ONLY)
  add_executable(CatLiveIntegration WIN32 Source/WindowsMain.cpp "${CMAKE_CURRENT_BINARY_DIR}/Windows.rc")
elseif(APPLE)
  target_sources(IntegrationCore PRIVATE Source/PlatformMac.mm)
  target_compile_options(IntegrationCore PRIVATE -fobjc-arc)
  target_link_libraries(IntegrationCore PUBLIC "-framework Foundation" "-framework CoreMIDI" "-framework SystemConfiguration" "-framework Security")
  add_executable(CatLiveIntegration MACOSX_BUNDLE Source/MacMain.mm)
  target_compile_options(CatLiveIntegration PRIVATE -fobjc-arc)
  add_custom_command(TARGET CatLiveIntegration POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:CatLiveIntegration>/Contents/Resources/Licenses"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/httplib/LICENSE" "$<TARGET_BUNDLE_DIR:CatLiveIntegration>/Contents/Resources/Licenses/cpp-httplib.txt"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/Resources/nlohmann-json-LICENSE.txt" "$<TARGET_BUNDLE_DIR:CatLiveIntegration>/Contents/Resources/Licenses/nlohmann-json.txt")
  target_link_libraries(CatLiveIntegration PRIVATE "-framework AppKit")
  set_target_properties(CatLiveIntegration PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Resources/Info.plist.in" MACOSX_BUNDLE_GUI_IDENTIFIER com.hookdeveloper.catlive.integration MACOSX_BUNDLE_BUNDLE_NAME "CatLive Integration" MACOSX_BUNDLE_SHORT_VERSION_STRING "1.28" MACOSX_BUNDLE_BUNDLE_VERSION "1" OUTPUT_NAME "CatLive Integration")
else()
  message(FATAL_ERROR "CatLive Integration supports macOS and Windows")
endif()
target_link_libraries(CatLiveIntegration PRIVATE IntegrationCore)
if(MSVC)
  set_property(TARGET CatLiveIntegration PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
add_executable(IntegrationTests Tests/IntegrationTests.cpp Tests/OutputTests.cpp Tests/GrandMA2Tests.cpp)
if(APPLE)
  target_sources(IntegrationTests PRIVATE Tests/MacMIDITests.mm)
endif()
target_include_directories(IntegrationTests PRIVATE ThirdParty/httplib)
target_link_libraries(IntegrationTests PRIVATE IntegrationCore)
if(MSVC)
  set_property(TARGET IntegrationTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_test(NAME IntegrationProtocol COMMAND IntegrationTests)
set_tests_properties(IntegrationProtocol PROPERTIES TIMEOUT 45)

install(TARGETS CatLiveIntegration BUNDLE DESTINATION . RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)
install(FILES ThirdParty/httplib/LICENSE DESTINATION Licenses RENAME cpp-httplib.txt)
install(FILES Resources/nlohmann-json-LICENSE.txt DESTINATION Licenses)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_VERSION "1.28")
set(CPACK_PACKAGE_FILE_NAME "CatLive-Integration-1.28-${CMAKE_SYSTEM_NAME}")
include(CPack)
